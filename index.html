/******************************************************************* * CONTROLE DE PRODUÇÃO — FEIRA · Aluminas Móveis / PCP * Backend Apps Script (Web App) para o painel HTML. * * Ajustado para o layout real da aba "Produtos": * - preenche a coluna ID sozinho (ela estava vazia — era por isso * que nada gravava: toda escrita procura a linha pelo ID) * - respeita PRIORIDADE como caixa de seleção (grava TRUE/FALSE) * - grava os setores com acento: LANÇADO / LANÇAR / NÃO PASSA / FALTA * - nunca escreve na coluna STATUS (é fórmula sua, fica intocada) * - grava também por GET (JSONP), então o painel finalmente * consegue ver o erro quando algo dá errado * * PUBLICAR: * Extensões > Apps Script > cole tudo > Salvar * Implantar > Gerenciar implantações > editar (lápis) * > Versão: Nova versão > Implantar * A URL /exec continua a mesma. *******************************************************************/ var SHEET_NAME = 'Produtos'; var TZ = 'America/Sao_Paulo'; var CONF_TEXTO = 'CONFERIDO'; // texto gravado na coluna CONFERENCIA var SECTORS = ['DESENHO','FUSAO','TUBULAR','PINTURA','ATELIE','TECELAGEM','MARCENARIA','MONTAGEM','EMBALAGEM']; /* como o valor é GRAVADO na planilha (com acento, igual você já usa) */ var SECTOR_WRITE = { '': '', 'FALTA': 'FALTA', 'LANCAR': 'LANÇAR', 'LANCADO': 'LANÇADO', 'NAO_PASSA': 'NÃO PASSA' }; /* colunas: nome canônico -> apelidos aceitos no cabeçalho da linha 1 */ var COLUMNS = { ID: ['ID'], MODELO: ['MODELO','PRODUTO','NOME'], DESENHO: ['DESENHO'], FUSAO: ['FUSAO','FUSÃO'], TUBULAR: ['TUBULAR'], PINTURA: ['PINTURA'], ATELIE: ['ATELIE','ATELIÊ'], TECELAGEM: ['TECELAGEM'], MARCENARIA: ['MARCENARIA'], MONTAGEM: ['MONTAGEM'], EMBALAGEM: ['EMBALAGEM'], CONFERENCIA: ['CONFERENCIA','CONFERÊNCIA','CONFERIDO','CONF'], DATA_FINALIZACAO: ['DATA_FINALIZACAO','DATA FINALIZACAO','DATA FINALIZAÇÃO','FINALIZACAO','DATA'], DATA_FINALIZACAO_TS: ['DATA_FINALIZACAO_TS','DATAFINALIZACAOTS','TS_FINALIZACAO'], PRIORIDADE: ['PRIORIDADE'], NAO_PRODUZ: ['NAO_PRODUZ','NAOPRODUZ','NAO_VAI_PRODUZIR','NÃO VAI PRODUZIR'] }; /* ---------------------------------------------------------------- */ /* infra */ /* ---------------------------------------------------------------- */ function getSheet_(){ var ss = SpreadsheetApp.getActiveSpreadsheet(); var sh = ss.getSheetByName(SHEET_NAME); if (!sh) throw new Error('Aba "' + SHEET_NAME + '" não encontrada. Ajuste SHEET_NAME no topo do script.'); return sh; } function norm_(v){ return String(v == null ? '' : v) .normalize('NFD').replace(/[\u0300-\u036f]/g,'') .toUpperCase().trim().replace(/[^A-Z0-9]+/g,'_').replace(/^_+|_+$/g,''); } function getSchema_(){ var sh = getSheet_(); var lastCol = Math.max(1, sh.getLastColumn()); var header = sh.getRange(1, 1, 1, lastCol).getValues()[0]; var found = {}; for (var c = 0; c < header.length; c++){ var h = norm_(header[c]); if (!h) continue; for (var canon in COLUMNS){ if (found[canon]) continue; var alias = COLUMNS[canon]; for (var a = 0; a < alias.length; a++){ if (norm_(alias[a]) === h){ found[canon] = c + 1; break; } } } } var created = false; for (var canon2 in COLUMNS){ if (!found[canon2]){ lastCol++; sh.getRange(1, lastCol).setValue(canon2); found[canon2] = lastCol; created = true; } } if (created) SpreadsheetApp.flush(); var sc = { sheet: sh, idx: found, lastCol: Math.max(lastCol, sh.getLastColumn()) }; ensureIds_(sc); return sc; } /* ESTE É O CONSERTO PRINCIPAL: toda linha que tem MODELO precisa ter um ID, senão nenhuma gravação consegue localizar a linha. Preenche só o que estiver faltando. */ function ensureIds_(sc){ var sh = sc.sheet; var last = sh.getLastRow(); if (last < 2) return; var n = last - 1; var ids = sh.getRange(2, sc.idx.ID, n, 1).getValues(); var mods = sh.getRange(2, sc.idx.MODELO, n, 1).getValues(); var maxId = 0; for (var i = 0; i < n; i++){ var v = Number(ids[i][0]); if (!isNaN(v) && v > maxId) maxId = v; } var mudou = false; for (var j = 0; j < n; j++){ var temModelo = String(mods[j][0] || '').trim() !== ''; var temId = String(ids[j][0] || '').trim() !== ''; if (temModelo && !temId){ ids[j][0] = ++maxId; mudou = true; } else if (!temModelo && temId){ ids[j][0] = ''; mudou = true; } } if (mudou) sh.getRange(2, sc.idx.ID, n, 1).setValues(ids); } function toBool_(v){ if (v === true) return true; if (v === false || v == null || v === '') return false; var s = norm_(v); return ['1','TRUE','VERDADEIRO','SIM','S','X'].indexOf(s) >= 0; } function isConferido_(v){ if (v === true) return true; if (v === false || v == null || v === '') return false; if (Object.prototype.toString.call(v) === '[object Date]') return true; var t = norm_(v); if (!t || t === '0' || t === 'FALSE' || t === 'NAO' || t === 'N') return false; return ['OK','TRUE','VERDADEIRO','1','SIM','S','X','CONFERIDO','CONFERIDA','FINALIZADO','CONCLUIDO'].indexOf(t) >= 0; } function toTs_(v){ if (v === '' || v == null) return null; if (Object.prototype.toString.call(v) === '[object Date]') return v.getTime(); var n = Number(v); return isNaN(n) || n <= 0 ? null : n; } /* data de finalização: usa o TS; se não houver, lê a coluna de data (Date ou texto dd/mm/aaaa), sempre travada ao meio-dia pra não escorregar de dia */ function resolveTs_(vTs, vData){ var ts = toTs_(vTs); if (ts) return ts; if (Object.prototype.toString.call(vData) === '[object Date]'){ return new Date(vData.getFullYear(), vData.getMonth(), vData.getDate(), 12, 0, 0).getTime(); } var m = String(vData == null ? '' : vData).trim().match(/^(\d{1,2})[\/\-\.](\d{1,2})[\/\-\.](\d{2,4})$/); if (m){ var ano = Number(m[3]); if (ano < 100) ano += 2000; return new Date(ano, Number(m[2]) - 1, Number(m[1]), 12, 0, 0).getTime(); } return null; } /* Date ao meio-dia no fuso certo — evita o deslocamento de um dia */ function dateFromTs_(ts){ var p = Utilities.formatDate(new Date(ts), TZ, 'yyyy-MM-dd').split('-'); return new Date(Number(p[0]), Number(p[1]) - 1, Number(p[2]), 12, 0, 0); } function findRowById_(sc, id){ var sh = sc.sheet; var last = sh.getLastRow(); if (last < 2) return -1; var ids = sh.getRange(2, sc.idx.ID, last - 1, 1).getValues(); var alvo = String(id).trim(); for (var i = 0; i < ids.length; i++){ if (String(ids[i][0]).trim() === alvo) return i + 2; } return -1; } /* grava flag respeitando caixa de seleção */ function setFlag_(rng, on, textoLigado){ var atual = rng.getValue(); if (typeof atual === 'boolean') rng.setValue(!!on); else rng.setValue(on ? (textoLigado || 1) : ''); } /* ---------------------------------------------------------------- */ /* LEITURA */ /* ---------------------------------------------------------------- */ function readRows_(){ var sc = getSchema_(); var sh = sc.sheet; var last = sh.getLastRow(); if (last < 2) return []; var values = sh.getRange(2, 1, last - 1, sc.lastCol).getValues(); var i = sc.idx; var out = []; for (var r = 0; r < values.length; r++){ var v = values[r]; if (String(v[i.MODELO - 1] || '').trim() === '') continue; // pula linhas vazias var obj = { id: v[i.ID - 1], modelo: v[i.MODELO - 1], conferencia: isConferido_(v[i.CONFERENCIA - 1]) ? 'OK' : '', dataFinalizacaoTs: resolveTs_(v[i.DATA_FINALIZACAO_TS - 1], v[i.DATA_FINALIZACAO - 1]), prioridade: toBool_(v[i.PRIORIDADE - 1]) ? 1 : 0, naoProduz: toBool_(v[i.NAO_PRODUZ - 1]) ? 1 : 0 }; for (var s = 0; s < SECTORS.length; s++){ obj[SECTORS[s]] = norm_(v[i[SECTORS[s]] - 1]); } out.push(obj); } return out; } /* ---------------------------------------------------------------- */ /* ENTRADAS: GET (leitura e escrita via JSONP) e POST (bulkSync) */ /* ---------------------------------------------------------------- */ function doGet(e){ var p = (e && e.parameter) || {}; var payload; try { if (p.action){ var lock = LockService.getScriptLock(); lock.waitLock(25000); try { payload = handleAction_(p); } finally { lock.releaseLock(); } } else { payload = { ok: true, rows: readRows_() }; } } catch (err){ payload = { ok: false, error: String(err && err.message ? err.message : err) }; } var json = JSON.stringify(payload); if (p.callback){ return ContentService.createTextOutput(p.callback + '(' + json + ')') .setMimeType(ContentService.MimeType.JAVASCRIPT); } return ContentService.createTextOutput(json).setMimeType(ContentService.MimeType.JSON); } function doPost(e){ var lock = LockService.getScriptLock(); try { lock.waitLock(25000); } catch (err){ return ContentService.createTextOutput(JSON.stringify({ok:false, error:'ocupado'})) .setMimeType(ContentService.MimeType.JSON); } var res; try { res = handleAction_(JSON.parse(e.postData.contents)); } catch (err){ res = { ok:false, error: String(err && err.message ? err.message : err) }; } finally { lock.releaseLock(); } return ContentService.createTextOutput(JSON.stringify(res)) .setMimeType(ContentService.MimeType.JSON); } function handleAction_(body){ var sc = getSchema_(); var a = body.action; if (a === 'updateCell') return actUpdateCell_(sc, body); if (a === 'updateModelo') return actUpdateModelo_(sc, body); if (a === 'updateConferencia') return actUpdateConferencia_(sc, body); if (a === 'updatePrioridade') return actUpdatePrioridade_(sc, body); if (a === 'updateNaoProduz') return actUpdateNaoProduz_(sc, body); if (a === 'addRow') return actAddRow_(sc, body); if (a === 'deleteRow') return actDeleteRow_(sc, body); if (a === 'bulkSync') return actBulkSync_(sc, body); return { ok:false, error:'ação desconhecida: ' + a }; } /* ---------------------------------------------------------------- */ /* AÇÕES */ /* ---------------------------------------------------------------- */ function actUpdateCell_(sc, body){ var setor = norm_(body.sector); if (SECTORS.indexOf(setor) === -1) return { ok:false, error:'setor inválido: ' + body.sector }; var r = findRowById_(sc, body.row); if (r < 0) return { ok:false, error:'não achei a linha do ID ' + body.row }; var valor = norm_(body.value); if (!(valor in SECTOR_WRITE)) return { ok:false, error:'valor inválido: ' + body.value }; sc.sheet.getRange(r, sc.idx[setor]).setValue(SECTOR_WRITE[valor]); return { ok:true }; } function actUpdateModelo_(sc, body){ var r = findRowById_(sc, body.row); if (r < 0) return { ok:false, error:'não achei a linha do ID ' + body.row }; sc.sheet.getRange(r, sc.idx.MODELO).setValue(String(body.value || '').trim()); return { ok:true }; } function actUpdateConferencia_(sc, body){ var r = findRowById_(sc, body.row); if (r < 0) return { ok:false, error:'não achei a linha do ID ' + body.row }; var on = String(body.conferencia || '').toUpperCase() === 'OK'; var ts = toTs_(body.dataFinalizacaoTs); setFlag_(sc.sheet.getRange(r, sc.idx.CONFERENCIA), on, CONF_TEXTO); sc.sheet.getRange(r, sc.idx.DATA_FINALIZACAO_TS).setValue(on && ts ? ts : ''); sc.sheet.getRange(r, sc.idx.DATA_FINALIZACAO).setValue(on && ts ? dateFromTs_(ts) : ''); return { ok:true }; } function actUpdatePrioridade_(sc, body){ var r = findRowById_(sc, body.row); if (r < 0) return { ok:false, error:'não achei a linha do ID ' + body.row }; setFlag_(sc.sheet.getRange(r, sc.idx.PRIORIDADE), toBool_(body.prioridade), 'SIM'); return { ok:true }; } function actUpdateNaoProduz_(sc, body){ var r = findRowById_(sc, body.row); if (r < 0) return { ok:false, error:'não achei a linha do ID ' + body.row }; var on = toBool_(body.naoProduz); setFlag_(sc.sheet.getRange(r, sc.idx.NAO_PRODUZ), on, 'SIM'); if (on) setFlag_(sc.sheet.getRange(r, sc.idx.PRIORIDADE), false, 'SIM'); return { ok:true }; } /* usa a primeira linha vazia dentro do intervalo; só então cresce a planilha. Escreve APENAS ID e MODELO — não encosta na coluna STATUS (fórmula sua). */ function actAddRow_(sc, body){ var sh = sc.sheet; var last = sh.getLastRow(); var alvo = -1, maxId = 0; if (last >= 2){ var mods = sh.getRange(2, sc.idx.MODELO, last - 1, 1).getValues(); var ids = sh.getRange(2, sc.idx.ID, last - 1, 1).getValues(); for (var i = 0; i < ids.length; i++){ var n = Number(ids[i][0]); if (!isNaN(n) && n > maxId) maxId = n; if (alvo < 0 && String(mods[i][0] || '').trim() === '') alvo = i + 2; } } if (alvo < 0) alvo = Math.max(last, 1) + 1; sh.getRange(alvo, sc.idx.ID).setValue(maxId + 1); sh.getRange(alvo, sc.idx.MODELO).setValue(String(body.modelo || '').trim()); return { ok:true, id: maxId + 1, linha: alvo }; } function actDeleteRow_(sc, body){ var r = findRowById_(sc, body.row); if (r < 0) return { ok:false, error:'não achei a linha do ID ' + body.row }; sc.sheet.deleteRow(r); return { ok:true }; } /* "Restaurar original" — reescreve só as colunas que o script conhece. STATUS e qualquer outra coluna sua ficam intactas. */ function actBulkSync_(sc, body){ var sh = sc.sheet; var incoming = body.rows || []; var last = sh.getLastRow(); var i = sc.idx; var alvos = [i.MODELO, i.CONFERENCIA, i.DATA_FINALIZACAO, i.DATA_FINALIZACAO_TS, i.PRIORIDADE, i.NAO_PRODUZ, i.ID]; for (var s = 0; s < SECTORS.length; s++) alvos.push(i[SECTORS[s]]); if (last >= 2){ for (var c = 0; c < alvos.length; c++){ sh.getRange(2, alvos[c], last - 1, 1).clearContent(); } } if (!incoming.length) return { ok:true, count:0 }; function col(idx, fn){ var out = []; for (var k = 0; k < incoming.length; k++) out.push([fn(incoming[k], k)]); sh.getRange(2, idx, out.length, 1).setValues(out); } col(i.ID, function(src,k){ return src.id != null ? src.id : (k+1); }); col(i.MODELO, function(src){ return src.modelo || ''; }); for (var s2 = 0; s2 < SECTORS.length; s2++){ (function(sec){ col(i[sec], function(src){ var v = norm_(src[sec]); return (v in SECTOR_WRITE) ? SECTOR_WRITE[v] : ''; }); })(SECTORS[s2]); } col(i.CONFERENCIA, function(src){ return isConferido_(src.conferencia) ? CONF_TEXTO : ''; }); col(i.DATA_FINALIZACAO_TS, function(src){ var t = toTs_(src.dataFinalizacaoTs); return t ? t : ''; }); col(i.DATA_FINALIZACAO, function(src){ var t = toTs_(src.dataFinalizacaoTs); return t ? dateFromTs_(t) : ''; }); col(i.PRIORIDADE, function(src){ return toBool_(src.prioridade); }); col(i.NAO_PRODUZ, function(src){ return toBool_(src.naoProduz) ? 'SIM' : ''; }); return { ok:true, count: incoming.length }; } /* ---------------------------------------------------------------- */ /* DIAGNÓSTICO — rode aqui no editor e veja "Registro de execução" */ /* ---------------------------------------------------------------- */ function diagnostico(){ var sc = getSchema_(); var sh = sc.sheet; Logger.log('Aba: ' + sh.getName() + ' | linhas: ' + sh.getLastRow() + ' | colunas: ' + sc.lastCol); Logger.log('Cabeçalho: ' + JSON.stringify(sh.getRange(1,1,1,sc.lastCol).getValues()[0])); Logger.log('Colunas reconhecidas: ' + JSON.stringify(sc.idx)); var rows = readRows_(); Logger.log('Modelos lidos: ' + rows.length); Logger.log('Com ID: ' + rows.filter(function(x){ return String(x.id||'').trim() !== ''; }).length); Logger.log('Conferidos: ' + rows.filter(function(x){ return x.conferencia === 'OK'; }).length); Logger.log('Prioridades: ' + rows.filter(function(x){ return x.prioridade; }).length); Logger.log('Não produz: ' + rows.filter(function(x){ return x.naoProduz; }).length); for (var k = 0; k < Math.min(3, rows.length); k++) Logger.log(JSON.stringify(rows[k])); } /* teste de gravação de verdade: escreve e desfaz na primeira linha */ function testeGravacao(){ var sc = getSchema_(); var rows = readRows_(); if (!rows.length) { Logger.log('Sem linhas.'); return; } var id = rows[0].id; Logger.log('Testando no ID ' + id + ' (' + rows[0].modelo + ')'); Logger.log('linha encontrada: ' + findRowById_(sc, id)); Logger.log(JSON.stringify(actUpdateNaoProduz_(sc, {row:id, naoProduz:1}))); SpreadsheetApp.flush(); Logger.log('agora naoProduz = ' + readRows_()[0].naoProduz + ' (esperado 1)'); Logger.log(JSON.stringify(actUpdateNaoProduz_(sc, {row:id, naoProduz:0}))); Logger.log('desfeito.'); }
